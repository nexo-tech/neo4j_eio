neo4j_eio User Guide

Path Traversal

This guide covers working with paths in Cypher using the current API: variable‑length patterns, shortest paths, all shortest paths, pattern comprehension, and relationship‑property reductions.

Variable‑length paths

let label = Printf.sprintf "Person_%d" (Random.int 1_000_000) in
let _ = Query_builder.execute_unit
  (Query_builder.raw (Printf.sprintf
     "CREATE (a:%s {name: 'Alice'})-[:KNOWS]->(b:%s {name: 'Bob'})-[:KNOWS]->(c:%s {name: 'Charlie'})-[:KNOWS]->(d:%s {name: 'David'})"
     label label label label))
  session in
let res = Query_builder.execute
  (Query_builder.raw (Printf.sprintf
     "MATCH (start:%s {name: 'Alice'})-[:KNOWS*1..3]->(connected:%s) RETURN DISTINCT connected.name AS name ORDER BY name"
     label label))
  session

Shortest path

let label = Printf.sprintf "City_%d" (Random.int 1_000_000) in
let _ = Query_builder.execute_unit
  (Query_builder.raw (Printf.sprintf
     "CREATE (sf:%s {name: 'San Francisco'}),(la:%s {name:'Los Angeles'}),(vegas:%s {name:'Las Vegas'}),(phx:%s {name:'Phoenix'}),
      (sf)-[:ROUTE {distance:380}]->(la),(sf)-[:ROUTE {distance:570}]->(vegas),(vegas)-[:ROUTE {distance:420}]->(la),(vegas)-[:ROUTE {distance:300}]->(phx),(la)-[:ROUTE {distance:370}]->(phx)"
     label label label label)) session in
let res = Query_builder.execute
  (Query_builder.raw (Printf.sprintf
     "MATCH (start:%s {name: 'San Francisco'}), (end:%s {name: 'Phoenix'}), p = shortestPath((start)-[:ROUTE*]-(end)) RETURN [n IN nodes(p) | n.name] AS cities, length(p) AS hops"
     label label)) session
(* Decode with Record.at_list Record.exact_text "cities" and Record.at_int "hops" *)

All shortest paths

let label = Printf.sprintf "Node_%d" (Random.int 1_000_000) in
let _ = Query_builder.execute_unit
  (Query_builder.raw (Printf.sprintf
     "CREATE (a:%s {name:'A'}),(b:%s {name:'B'}),(c:%s {name:'C'}),(d:%s {name:'D'}),(a)-[:CONNECTED]->(b)-[:CONNECTED]->(d),(a)-[:CONNECTED]->(c)-[:CONNECTED]->(d)"
     label label label label)) session in
let res = Query_builder.execute
  (Query_builder.raw (Printf.sprintf
     "MATCH (start:%s {name:'A'}), (end:%s {name:'D'}), p = allShortestPaths((start)-[:CONNECTED*]-(end)) RETURN [n IN nodes(p) | n.name] AS path"
     label label)) session
(* Each row is a path (list of names), decode via Record.at_list Record.exact_text "path" *)

Pattern comprehension

let label = Printf.sprintf "User_%d" (Random.int 1_000_000) in
let _ = Query_builder.execute_unit
  (Query_builder.raw (Printf.sprintf
     "CREATE (alice:%s {name:'Alice'}),(bob:%s {name:'Bob'}),(p1:Project {name:'API'}),(p2:Project {name:'UI'}),(p3:Project {name:'DB'}),(alice)-[:WORKS_ON]->(p1),(alice)-[:WORKS_ON]->(p2),(bob)-[:WORKS_ON]->(p2),(bob)-[:WORKS_ON]->(p3)"
     label label)) session in
let res = Query_builder.execute
  (Query_builder.raw (Printf.sprintf
     "MATCH (u:%s) RETURN u.name AS user, [(u)-[:WORKS_ON]->(p:Project) | p.name] AS projects"
     label)) session
(* Decode: Record.at_text "user", Record.at_list Record.exact_text "projects" *)

Relationship‑property reductions

let label = Printf.sprintf "Location_%d" (Random.int 1_000_000) in
let _ = Query_builder.execute_unit
  (Query_builder.raw (Printf.sprintf
     "CREATE (a:%s {name:'A'}),(b:%s {name:'B'}),(c:%s {name:'C'}),(d:%s {name:'D'}),
      (a)-[:PATH {cost:1}]->(b),(b)-[:PATH {cost:1}]->(d),(a)-[:PATH {cost:2}]->(c),(c)-[:PATH {cost:2}]->(d)"
     label label label label)) session in
let res = Query_builder.execute
  (Query_builder.raw (Printf.sprintf
     "MATCH p=(start:%s {name:'A'})-[:PATH*]->(end:%s {name:'D'}) RETURN [n IN nodes(p) | n.name] AS path, reduce(total = 0, r IN relationships(p) | total + r.cost) AS total_cost ORDER BY total_cost LIMIT 1"
     label label)) session
(* Decode: Record.at_list Record.exact_text "path", Record.at_int "total_cost" *)

Decoding paths as Value.Path

  • Instead of projecting lists, you can return the path p and decode with Record.at_path "p" to get Value.path with path_nodes and path_rels (unbound relationships).

Tips

  • Use projected lists ([n IN nodes(p) | ...]) when you only need certain attributes.
  • Use length(p), nodes(p), relationships(p) in Cypher for concise path analysis.
  • Alias fields (AS ...) to ensure stable keys for decoding.
